18inline std::vector<std::string> parseCsvRow(const std::string& line) {
19 std::vector<std::string> fields;
20 std::string current;
21 bool inQuotes = false;
22
23 for (std::size_t i = 0; i < line.size(); ++i) {
24 const char ch = line[i];
25
26 if (ch == '"') {
27 if (inQuotes && i + 1 < line.size() && line[i + 1] == '"') {
28 current.push_back('"');
29 ++i;
30 } else {
31 inQuotes = !inQuotes;
32 }
33 } else if (ch == ',' && !inQuotes) {
34 fields.push_back(current);
35 current.clear();
36 } else {
37 current.push_back(ch);
38 }
39 }
40
41 if (inQuotes) {
42 throw std::runtime_error("Malformed CSV row: unmatched quote");
43 }
44
45 fields.push_back(current);
46 return fields;
47}

52inline std::string escapeCsvField(const std::string& field) {
53 const bool needsQuotes = field.find_first_of(",\"\n\r") != std::string::npos;
54 if (!needsQuotes) {
55 return field;
56 }
57
58 std::string escaped;
59 escaped.reserve(field.size() + 2);
60 escaped.push_back('"');
61 for (const char ch : field) {
62 if (ch == '"') {
63 escaped += "\"\"";
64 } else {
65 escaped.push_back(ch);
66 }
67 }
68 escaped.push_back('"');
69 return escaped;
70}
